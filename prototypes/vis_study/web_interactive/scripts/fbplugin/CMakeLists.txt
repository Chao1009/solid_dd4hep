# Minimal build of Firebird's DD4hep trajectory writer (eic/firebird dd4hep-plugin, event action only)
cmake_minimum_required(VERSION 3.16)
project(solid_firebird_writer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT FB_SRC)
  message(FATAL_ERROR "pass -DFB_SRC=<dir with FirebirdTrajectoryWriterEventAction.cpp>")
endif()
find_package(DD4hep REQUIRED COMPONENTS DDCore DDG4)
find_package(Geant4 REQUIRED)
find_package(fmt REQUIRED)
dd4hep_configure_output()
dd4hep_set_compiler_flags()
dd4hep_add_plugin(firebird_writer SOURCES ${FB_SRC}/FirebirdTrajectoryWriterEventAction.cpp
                  USES DD4hep::DDCore DD4hep::DDG4 fmt::fmt Geant4::Interface)
